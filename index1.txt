      // Nav highlight
      const sections = document.querySelectorAll("section");

      window.addEventListener("scroll", () => {
        let cur = "";
        sections.forEach((s) => {
          if (pageYOffset >= s.offsetTop - 120) cur = s.id;
        });
        navLinks.forEach((l) => {
          l.classList.remove("nav-active");
          if (l.getAttribute("href").includes(cur))
            l.classList.add("nav-active");
        });
      });

      // Typewriter
      const words = [
        "Frontend Developer",
        "React Enthusiast",
        "Problem Solver",
        "Tech Learner",
      ];
      let i = 0,
        j = 0,
        del = false;
      function type() {
        const el = document.getElementById("typewriter");
        if (!el) return;
        const w = words[i];
        if (del) {
          el.innerHTML = w.substring(0, --j);
          if (!j) {
            del = false;
            i = (i + 1) % words.length;
            setTimeout(type, 120);
            return;
          }
          setTimeout(type, 40);
        } else {
          el.innerHTML = w.substring(0, ++j);
          if (j === w.length) {
            del = true;
            setTimeout(type, 1600);
            return;
          }
          setTimeout(type, 70);
        }
      }
      document.addEventListener("DOMContentLoaded", type);

      // Reveal
      const reveals = document.querySelectorAll(".reveal");
      new IntersectionObserver(
        (e) =>
          e.forEach((en) => {
            if (en.isIntersecting) en.target.classList.add("in");
          }),
        { threshold: 0.1 },
      ).observe &&
        (() => {
          const io = new IntersectionObserver(
            (entries) =>
              entries.forEach((e) => {
                if (e.isIntersecting) e.target.classList.add("in");
              }),
            { threshold: 0.1 },
          );
          reveals.forEach((r) => io.observe(r));
        })();

      // Scroll top
      const scrollBtn = document.getElementById("scrollTopBtn");
      window.addEventListener("scroll", () => {
        if (window.scrollY > 500) scrollBtn.classList.remove("hidden");
        else scrollBtn.classList.add("hidden");

    <script>
      (function () {
        const cv = document.getElementById("global-canvas");
        const cx = cv.getContext("2d");
        let pts = [];
        const r = () => {
          cv.width = innerWidth;
          cv.height = innerHeight;
        };
        window.addEventListener("resize", r);
        r();
        const m = { x: 0, y: 0 };
        window.addEventListener("mousemove", (e) => {
          m.x = e.x;
          m.y = e.y;
          for (let i = 0; i < 4; i++)
            pts.push({
              x: m.x,
              y: m.y,
              s: Math.random() * 2 + 0.4,
              vx: (Math.random() - 0.5) * 1.4,
              vy: (Math.random() - 0.5) * 1.4,
              l: 1,
            });
        });
        class P {
          constructor(d) {
            Object.assign(this, d);
          }
        }
        function draw() {
          cx.clearRect(0, 0, cv.width, cv.height);
          pts = pts.filter((p) => p.l > 0 && p.s > 0);
          pts.forEach((p) => {
            p.x += p.vx;
            p.y += p.vy;
            p.l -= 0.028;
            p.s -= 0.018;
            cx.beginPath();
            cx.arc(p.x, p.y, Math.max(0, p.s), 0, Math.PI * 2);
            cx.fillStyle = `rgba(245,158,11,${p.l * 0.5})`;
            cx.fill();
          });
          requestAnimationFrame(draw);
        }
        draw();
      })();
    </script>

        cv.width = home.offsetWidth;
        cv.height = home.offsetHeight;
        const mouse = { x: null, y: null };
        window.addEventListener("mousemove", (e) => {
          const rect = cv.getBoundingClientRect();
          mouse.x =
            e.clientY >= rect.top && e.clientY <= rect.bottom
              ? e.clientX - rect.left
              : null;
          mouse.y =
            e.clientY >= rect.top && e.clientY <= rect.bottom
              ? e.clientY - rect.top
              : null;
        });
        function Dot() {
          this.x = Math.random() * cv.width;
          this.y = Math.random() * cv.height;
          this.dx = (Math.random() - 0.5) * 0.7;
          this.dy = (Math.random() - 0.5) * 0.7;
          this.s = Math.random() * 1.3 + 0.4;
        }
        Dot.prototype.update = function () {
          if (this.x < 0 || this.x > cv.width) this.dx *= -1;
          if (this.y < 0 || this.y > cv.height) this.dy *= -1;
          if (mouse.x) {
            const dx = mouse.x - this.x,
              dy = mouse.y - this.y,
              d = Math.sqrt(dx * dx + dy * dy);
            if (d < 100) {
              if (mouse.x < this.x && this.x < cv.width - this.s * 10)
                this.x += 3;
              if (mouse.x > this.x && this.x > this.s * 10) this.x -= 3;
              if (mouse.y < this.y && this.y < cv.height - this.s * 10)
                this.y += 3;
              if (mouse.y > this.y && this.y > this.s * 10) this.y -= 3;
            }
          }
          this.x += this.dx;
          this.y += this.dy;
          cx.beginPath();
          cx.arc(this.x, this.y, this.s, 0, Math.PI * 2);
          cx.fillStyle = "rgba(245,158,11,0.3)";
          cx.fill();
        };
        function init() {
          pts = [];
          const n = (cv.width * cv.height) / 8000;
          for (let i = 0; i < n; i++) pts.push(new Dot());
        }
        function connect() {
          for (let a = 0; a < pts.length; a++)
            for (let b = a + 1; b < pts.length; b++) {
              const d = (pts[a].x - pts[b].x) ** 2 + (pts[a].y - pts[b].y) ** 2;
              const max = (cv.width / 7) * (cv.height / 7);
              if (d < max) {
                cx.strokeStyle = `rgba(245,158,11,${(1 - d / max) * 0.2})`;
                cx.lineWidth = 0.6;
                cx.beginPath();
                cx.moveTo(pts[a].x, pts[a].y);
                cx.lineTo(pts[b].x, pts[b].y);
                cx.stroke();
              }
            }
        }
        function anim() {
          requestAnimationFrame(anim);
          cx.clearRect(0, 0, cv.width, cv.height);
          pts.forEach((p) => p.update());
          connect();
        }
        init();
        anim();
      })();
    </script>
  </body>
